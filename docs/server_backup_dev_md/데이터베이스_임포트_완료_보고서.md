### 거리 기반 검색
```sql
-- 특정 좌표(강남역)에서 5km 이내 병원 검색
SELECT
  id, name, address, rating,
  (6371 * acos(
    cos(radians(37.498095)) * cos(radians(latitude)) *
    cos(radians(longitude) - radians(127.027610)) +
    sin(radians(37.498095)) * sin(radians(latitude))
  )) AS distance_km
FROM gangnamunni_hospitals
WHERE latitude IS NOT NULL AND longitude IS NOT NULL
HAVING distance_km < 5
ORDER BY distance_km;
```

    $query = DB::table('gangnamunni_hospitals');

    if ($request->category) {
        $query->whereJsonContains('categories', $request->category);
    }

    if ($request->min_rating) {
        $query->where('rating', '>=', $request->min_rating);

    return $query->orderBy('rating', 'desc')->paginate(20);
