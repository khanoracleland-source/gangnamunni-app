## ⚠️ DB 관리 전략

### 매일 자동 백업 스크립트
```php
// H:\carebaby_project\server_backup_dev\hairwang_dev_sns_shop_2026_01_18\gangnamunni\backup_db.php

<?php
$host = '124.111.85.242';
$user = 'root';
$pass = '<REMOVED_DB_PASSWORD>';
$db = 'hairwang';
$backup_dir = __DIR__ . '/../backups/db';

// 백업 폴더 생성
if (!is_dir($backup_dir)) {
    mkdir($backup_dir, 0755, true);
}

// 백업 파일명 (날짜_시간)
$backup_file = $backup_dir . '/hairwang_' . date('Y-m-d_His') . '.sql';

// mysqldump 실행
$command = "mysqldump -h{$host} -u{$user} -p{$pass} {$db} > {$backup_file}";
exec($command, $output, $return);

if ($return === 0) {
    echo "✅ DB 백업 완료: {$backup_file}\n";
    echo "파일 크기: " . round(filesize($backup_file) / 1024 / 1024, 2) . " MB\n";

    // 7일 이상 된 백업 삭제
    $files = glob($backup_dir . '/*.sql');
    foreach ($files as $file) {
        if (time() - filemtime($file) > 7 * 24 * 60 * 60) {
            unlink($file);
            echo "🗑️  오래된 백업 삭제: " . basename($file) . "\n";
        }
    }
} else {
    echo "❌ DB 백업 실패\n";
    exit(1);
}
?>
```
