```sql
CREATE TABLE subscription_transactions (
```

**단계별 실행**:

```sql
-- Step 1: Plan 테이블 생성
CREATE TABLE IF NOT EXISTS `plan` (
    `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    `name` VARCHAR(100) NOT NULL,
    `type` VARCHAR(50) DEFAULT 'monthly',
    `duration` INT DEFAULT 1,
    `amount` DOUBLE DEFAULT 0,
    `identifier` VARCHAR(50) UNIQUE,
    `trial_period` INT DEFAULT 0,
    `planlimitation` TEXT,
    `description` TEXT,
    `status` TINYINT DEFAULT 1,
    `created_by` INT UNSIGNED,
    `updated_by` INT UNSIGNED,

    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

-- Step 2: Subscriptions 테이블 생성
CREATE TABLE IF NOT EXISTS `subscriptions` (
    `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    `plan_id` INT UNSIGNED,
    `user_id` INT UNSIGNED,
    `start_date` DATETIME,
    `end_date` DATETIME,
    `status` VARCHAR(50) DEFAULT 'active',
    `amount` DOUBLE DEFAULT 0,
    `name` VARCHAR(100),
    `identifier` VARCHAR(50),
    `type` VARCHAR(50),
    `duration` INT,
    `plan_type` VARCHAR(50),
    `payment_id` INT UNSIGNED,
    `created_by` INT UNSIGNED,
    `updated_by` INT UNSIGNED,
    `deleted_by` INT UNSIGNED,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    `deleted_at` TIMESTAMP NULL,
    INDEX `idx_user_id` (`user_id`),
    INDEX `idx_plan_id` (`plan_id`),
    INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Step 3: Subscription Transactions 테이블 생성
CREATE TABLE IF NOT EXISTS `subscription_transactions` (
    `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    `subscription_id` INT UNSIGNED,
    `transaction_type` VARCHAR(50) DEFAULT 'payment',
    `amount` DOUBLE DEFAULT 0,
    `payment_method` VARCHAR(50),
    `transaction_date` DATETIME,

    `deleted_by` INT UNSIGNED,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    `deleted_at` TIMESTAMP NULL,
    INDEX `idx_subscription_id` (`subscription_id`),
    INDEX `idx_transaction_type` (`transaction_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Step 4: Plan Limitation 테이블 생성
CREATE TABLE IF NOT EXISTS `plan_limitation` (
    `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    `name` VARCHAR(100) NOT NULL,
    `type` VARCHAR(50),
    `identifier` VARCHAR(50) UNIQUE,
    `description` TEXT,
    `status` TINYINT DEFAULT 1,
    `created_by` INT UNSIGNED,
    `updated_by` INT UNSIGNED,
    `deleted_by` INT UNSIGNED,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    `deleted_at` TIMESTAMP NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Step 5: Plan Limitation Mapping 테이블 생성
CREATE TABLE IF NOT EXISTS `plan_limitation_mapping` (
    `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    `plan_id` INT UNSIGNED,
    `limitation_id` INT UNSIGNED,
    `value` VARCHAR(255),
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX `idx_plan_id` (`plan_id`),
    INDEX `idx_limitation_id` (`limitation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```
