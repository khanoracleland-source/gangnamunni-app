```php
/**
 * 네이티브 공통 콜백
 * Android / iOS 모두 여기로 토큰을 전달
 */
function onTokenReceived(token) {
    if (!token) return;

    console.log('[앱 브리지] FCM Token 수신:', token);

    // 그누보드 표준 진입점 재사용
    if (window.receiveFCMToken) {
        window.receiveFCMToken(token);
    } else {
        // 혹시 아직 로드 전이면 로컬에만 저장
        console.warn('[앱 브리지] receiveFCMToken 함수 아직 로드 안됨, localStorage만 저장');
        localStorage.setItem('fcm_token', token);
    }
}

/**
 * 페이지 로드 후 토큰 요청
 */
document.addEventListener('DOMContentLoaded', function () {
    console.log('[앱 브리지] DOMContentLoaded - 토큰 요청 시작');

    // Android
    if (window.Android && typeof Android.requestToken === 'function') {
        console.log('[앱 브리지] Android 브리지 감지 - requestToken() 호출');
        Android.requestToken();
    } else {
        console.log('[앱 브리지] Android 브리지 없음');
    }

    // iOS
    if (
        window.webkit &&
        window.webkit.messageHandlers &&
        window.webkit.messageHandlers.invokeAction
    ) {
        console.log('[앱 브리지] iOS 브리지 감지 - getToken 요청');
        window.webkit.messageHandlers.invokeAction.postMessage('getToken');
    } else {
        console.log('[앱 브리지] iOS 브리지 없음');
    }
});
</script>
```

### 📍 Step 2: Android 앱 구현 (Kotlin)

```kotlin
// MainActivity.kt 또는 BaseWebViewActivity.kt
class MainActivity : AppCompatActivity() {

    private lateinit var webView: WebView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // WebView 설정
        webView = findViewById(R.id.webView)
        webView.settings.apply {
            javaScriptEnabled = true
            domStorageEnabled = true
        }

        // JavaScript 인터페이스 추가
        webView.addJavascriptInterface(WebAppInterface(this), "Android")

        // 웹 페이지 로드
        webView.loadUrl("https://yourdomain.com")
    }

    // JavaScript 인터페이스
    inner class WebAppInterface(private val context: Context) {

        @JavascriptInterface
        fun requestToken() {
            // FCM 토큰 가져오기
            FirebaseMessaging.getInstance().token.addOnSuccessListener { token ->
                Log.d("FCM", "Token: $token")

                // 웹뷰로 토큰 전달 (메인 스레드에서 실행)
                runOnUiThread {
                    webView.evaluateJavascript(
                        "onTokenReceived('$token')",
                        null
                    )
                }
            }.addOnFailureListener { e ->
                Log.e("FCM", "Failed to get token", e)
            }
        }
    }
}
```

### 📍 Step 3: iOS 앱 구현 (Swift)

```swift
// ViewController.swift
import UIKit
import WebKit
import FirebaseMessaging

class ViewController: UIViewController, WKScriptMessageHandler, WKNavigationDelegate {

    var webView: WKWebView!

    override func viewDidLoad() {
        super.viewDidLoad()

        // WKWebView 설정
        let config = WKWebViewConfiguration()
        let userContentController = WKUserContentController()

        // JavaScript 메시지 핸들러 등록
        userContentController.add(self, name: "invokeAction")
        config.userContentController = userContentController

        webView = WKWebView(frame: view.bounds, configuration: config)
        webView.navigationDelegate = self
        view.addSubview(webView)

        // 웹 페이지 로드
        if let url = URL(string: "https://yourdomain.com") {
            webView.load(URLRequest(url: url))
        }

        // FCM 토큰 갱신 리스너
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(tokenRefreshNotification),
            name: Notification.Name.MessagingRegistrationTokenRefreshed,
            object: nil
        )
    }

    // JavaScript에서 호출하는 메시지 핸들러
    func userContentController(_ userContentController: WKUserContentController, didReceive message: WKScriptMessage) {
        if message.name == "invokeAction", let action = message.body as? String {
            if action == "getToken" {
                sendTokenToWebView()
            }
        }
    }

    // FCM 토큰을 웹뷰로 전달
    func sendTokenToWebView() {
        Messaging.messaging().token { token, error in
            if let error = error {
                print("Error fetching FCM token: \(error)")
            } else if let token = token {
                print("FCM Token: \(token)")

                // 토큰 이스케이프 처리
                let escapedToken = token
                    .replacingOccurrences(of: "\\", with: "\\\\")
                    .replacingOccurrences(of: "'", with: "\\'")

                // 웹뷰로 토큰 전달
                let script = "onTokenReceived('\(escapedToken)')"
                self.webView.evaluateJavaScript(script) { result, error in
                    if let error = error {
                        print("Error sending token to WebView: \(error)")
                    } else {
                        print("Token sent to WebView successfully")
                    }
                }
            }
        }
    }

    // FCM 토큰 갱신 시 호출
    @objc func tokenRefreshNotification() {
        sendTokenToWebView()
    }

    // 페이지 로드 완료 시
    func webView(_ webView: WKWebView, didFinish navigation: WKNavigation!) {
        print("Page loaded")
        // 필요시 토큰 재전송
        sendTokenToWebView()
    }
}
```

## 3. 테이블 구조 및 저장 확인

### 📊 DB 테이블: g5_push_tokens

```sql
CREATE TABLE `g5_push_tokens` (
    `pt_id` int(11) NOT NULL AUTO_INCREMENT,
    `mb_id` varchar(20) NOT NULL DEFAULT '',           -- ✅ 회원 ID 저장됨
    `pt_token` varchar(500) NOT NULL,                  -- FCM 토큰
    `pt_platform` varchar(10) NOT NULL DEFAULT '',     -- android/ios
    `pt_device_id` varchar(100) NOT NULL DEFAULT '',
    `pt_app_version` varchar(20) NOT NULL DEFAULT '',
    `pt_device_info` text,
    `pt_datetime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `pt_update_datetime` datetime DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    `pt_use` tinyint(4) NOT NULL DEFAULT '1',
    PRIMARY KEY (`pt_id`),
    UNIQUE KEY `pt_token` (`pt_token`),                -- 토큰 중복 방지
    KEY `mb_id` (`mb_id`),
    KEY `pt_device_id` (`pt_device_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
```

### ✅ 회원 ID 저장 확인

**[bbs/ajax.push_token.php:23-31](bbs/ajax.push_token.php:23-31)**
```php
// 세션에서 회원 ID 자동 추출
```

## 8. 추가 개선 사항 (선택)

### 🔧 비로그인 토큰 저장 지원

현재는 로그인 필수인데, 비로그인도 지원하려면:

**[bbs/ajax.push_token.php](bbs/ajax.push_token.php) 수정**
```php
// 기존 (로그인 필수)
if(!$member['mb_id']) {
    die(json_encode(['success' => false, 'message' => '로그인 필요']));
}

// 수정 (비로그인 허용)
$mb_id = $member['mb_id'] ?? '';  // 비로그인은 빈 값

// INSERT 시 mb_id 조건부 업데이트
$sql = "INSERT INTO g5_push_tokens
        SET mb_id = '{$mb_id}',
            pt_token = '{$token}',
            pt_platform = '{$platform}',
            pt_datetime = NOW()
        ON DUPLICATE KEY UPDATE
            mb_id = IF('{$mb_id}' != '', '{$mb_id}', mb_id),  -- 로그인 시만 업데이트
            pt_datetime = NOW()";
```

**장점**:
- 비로그인 → 로그인 시 자동으로 mb_id 연결
- 앱 설치 후 즉시 토큰 저장 가능
