## 개요

본 보고서는 `sns_pg연동_개발.md`와 `AI_개발_프롬프트.md` 문서를 기반으로, 세 프로젝트(carebaby, frezka_admin_panel_v4.0.3, kivicare-laravel-web-v1.9.0)에서 SNS 로그인과 PG 결제 API를 통합 구현하기 위한 분석 결과를 정리한 것입니다.

**분석 대상 프로젝트:**
1. **carebaby**: PHP 기반 NFOR 솔루션 (쇼핑몰 + 예약 시스템)

**API 인증 미들웨어 (api/common.php):**
```php
<?php
// API 키 검증
function api_authenticate() {
    $api_key = $_SERVER['HTTP_API_KEY'] ?? '';
    $api_secret = $_SERVER['HTTP_API_SECRET'] ?? '';
    $request_ip = $_SERVER['REMOTE_ADDR'];

     // API 키 검증
     $api_key_info = sql_fetch("
         SELECT * FROM nfor_addon_sns_pg_api_keys
         WHERE ak_key = '{$api_key}'
         AND ak_secret = '{$api_secret}'
         AND ak_status =1
         AND (ak_expired_at IS NULL OR ak_expired_at > NOW())
     ");

    if (!$api_key_info) {
        api_response(false, 'INVALID_API_KEY', '유효하지 않은 API 키입니다.', [], 401);
    }

    // IP 화이트리스트 검증
    if (!api_verify_ip_whitelist($api_key_info['ak_id'], $request_ip)) {
        api_response(false, 'IP_NOT_ALLOWED', '허용되지 않은 IP입니다.', [], 403);
    }

     // 마지막 사용 시간 업데이트
     sql_query("
         UPDATE nfor_addon_sns_pg_api_keys
         SET ak_last_used = NOW()
         WHERE ak_id = {$api_key_info['ak_id']}
     ");

    return $api_key_info;
}

// IP 화이트리스트 검증
function api_verify_ip_whitelist($ak_id, $request_ip) {
    $ips = sql_query("
        SELECT ai_ip FROM nfor_addon_sns_pg_api_ips
        WHERE ak_id = {$ak_id}
        AND ai_status =1
    ");

    if (sql_num_rows($ips) == 0) {
        return true; // IP 제한 없음
    }

    while ($ip = sql_fetch_array($ips)) {
        if (ip_match($request_ip, $ip['ai_ip'])) {
            return true;
        }
    }

    return false;
}

// API 로그 기록
function api_log($ak_id, $endpoint, $method, $request_data, $response_code, $response_data, $error = null, $execution_time = 0) {
    sql_query("
        INSERT INTO nfor_addon_sns_pg_api_logs
        (ak_id, al_endpoint, al_method, al_request_ip, al_request_data, al_response_code, al_response_data, al_error, al_execution_time, al_created_at)
        VALUES
        ('{$ak_id}', '{$endpoint}', '{$method}', '{$_SERVER['REMOTE_ADDR']}', '" . json_encode($request_data) . "', '{$response_code}', '" . json_encode($response_data) . "', '" . json_encode($error) . "', '{$execution_time}', NOW())
    ");
}
?>
```

#### 3.3.3 API 키 인증 미들웨어

```php
<?php
// Modules/APIIntegration/Http/Middleware/ApiKeyAuth.php
namespace Modules\APIIntegration\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Modules\APIIntegration\Models\ApiKey;

class ApiKeyAuth
{
    public function handle(Request $request, Closure $next)
    {
        $api_key = $request->header('API-Key');
        $api_secret = $request->header('API-Secret');

        // API 키 검증
        $apiKey = ApiKey::where('ak_key', $api_key)
            ->where('ak_secret', $api_secret)
            ->where('ak_status', 1)
            ->where(function ($query) {
                $query->whereNull('ak_expired_at')
                    ->orWhere('ak_expired_at', '>', now());
            })
            ->first();

        if (!$apiKey) {
            return response()->json([
                'success' => false,
                'code' => 'INVALID_API_KEY',
                'message' => '유효하지 않은 API 키입니다.',
                'timestamp' => now()
            ], 401);
        }

        // 마지막 사용 시간 업데이트
        $apiKey->update(['ak_last_used' => now()]);

        // API 키를 요청에 저장
        $request->merge(['api_key' => $apiKey]);

        return $next($request);
    }
}
?>
```
